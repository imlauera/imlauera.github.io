// Compile: gcc curlcurl.c -o curlcurl -lcurl
#include <curl/curl.h>

CURL *curl = curl_easy_init();
curl_easy_setopt(curl, CURLOPT_URL, 
    "https://imlauera.github.io/index.xml");
curl_easy_perform(curl);
